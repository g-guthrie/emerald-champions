static bool32 PairWaitingChangesPosition(enum BattleSide side)
{
    // A one-turn HP score must not prefer indefinitely postponing the same
    // exchange. Be conservative about time-dependent positions: these may
    // justify spending both actions on guards, and retain ordinary scoring.
    if (gBattleWeather || gFieldStatuses || gFieldTimers.terrainTimer)
        return TRUE;
    for (u32 side = 0; side < NUM_BATTLE_SIDES; side++)
    {
        const struct SideTimer *timer = &gSideTimers[side];
        if (timer->reflectTimer || timer->lightscreenTimer || timer->mistTimer
         || timer->safeguardTimer || timer->auroraVeilTimer || timer->tailwindTimer
         || timer->luckyChantTimer || timer->retaliateTimer || timer->damageNonTypesTimer
         || timer->rainbowTimer || timer->seaOfFireTimer || timer->swampTimer)
            return TRUE;
    }
    for (enum BattlerId battler = 0; battler < gBattlersCount; battler++)
    {
        if (!IsBattlerAlive(battler))
            continue;
        const struct Volatiles *v = &gBattleMons[battler].volatiles;
        if (gBattleMons[battler].status1 || GetBattlerSecondaryDamage(battler)
         || gBattleStruct->wish[battler].counter || gBattleStruct->futureSight[battler].counter
         || v->perishSong || v->yawn || v->aquaRing || v->root || v->saltCure || v->octolock
         || v->confusionTimer || v->uproarTurns || v->bideTurns || v->multipleTurns
         || v->rechargeTimer || v->slowStartTimer || v->embargoTimer || v->magnetRiseTimer
         || v->telekinesisTimer || v->healBlockTimer || v->tauntTimer || v->tormentTimer
         || v->laserFocusTimer || v->throatChopTimer || v->syrupBombTimer
         || v->disabledMove || v->encoredMove || v->destinyBond || v->glaiveRush)
            return TRUE;
        switch (gAiLogicData->abilities[battler])
        {
        case ABILITY_TRUANT:
        case ABILITY_CUD_CHEW:
        case ABILITY_HARVEST:
        case ABILITY_MOODY:
        case ABILITY_PICKUP:
        case ABILITY_SPEED_BOOST:
        case ABILITY_POWER_CONSTRUCT:
        case ABILITY_HUNGER_SWITCH:
        case ABILITY_ZEN_MODE:
            return TRUE;
        default: break;
        }
        switch (gAiLogicData->holdEffects[battler])
        {
        case HOLD_EFFECT_TOXIC_ORB:
        case HOLD_EFFECT_FLAME_ORB:
        case HOLD_EFFECT_LEFTOVERS:
        case HOLD_EFFECT_BLACK_SLUDGE:
        case HOLD_EFFECT_STICKY_BARB:
            return TRUE;
        default: break;
        }
        for (u32 index = 0; index < MAX_MON_MOVES; index++)
        {
            enum Move move = gBattleMons[battler].moves[index];
            if (IsMoveUnusable(index, move, gAiLogicData->moveLimitations[battler]))
                continue;
            // Exhausting an opposing attack or waiting out first-turn-only
            // pressure can change the next exchange even on an empty field.
            if (GetBattlerSide(battler) != side
             && ((gBattleMons[battler].pp[index] == 1 && !IsBattleMoveStatus(move))
                 || (gBattleStruct->battlerState[battler].isFirstTurn && move == MOVE_FAKE_OUT)))
                return TRUE;
        }
    }
    return FALSE;
}
